cmake_minimum_required(VERSION 3.10)
project(DragonZip CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(DRAGONZIP_NATIVE "Shu protsessorga moslab yig'ish (-march=native)" OFF)

find_package(Threads REQUIRED)
add_executable(dragonzip dragonzip.cpp)
target_link_libraries(dragonzip PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(dragonzip PRIVATE /O2 /utf-8)
  if(DRAGONZIP_NATIVE)
    target_compile_options(dragonzip PRIVATE /arch:AVX2)
  endif()
else()
  if(DRAGONZIP_NATIVE)
    target_compile_options(dragonzip PRIVATE -march=native)
  endif()
endif()
if(WIN32 AND NOT MSVC)
  target_link_libraries(dragonzip PRIVATE shell32)
endif()
